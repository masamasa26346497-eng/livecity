# 提案（未適用）: .github/workflows/pr-fast-tests.yml の置き換え案
# GitHub App に workflow 書き込み権限が無いため、UI/権限のある資格情報で人手適用すること。
# 注意: workflow_dispatch は workflow がデフォルトブランチに存在しないと使えない。
name: PR fast tests
on:
  pull_request:
    branches: [main]
  workflow_dispatch:
    inputs:
      full:
        description: "全件テスト(npm test)を baseline 比較なしで実行"
        type: boolean
        default: false
permissions:
  contents: read
concurrency:
  group: pr-fast-${{ github.workflow }}-${{ github.event.pull_request.number || github.ref }}
  cancel-in-progress: true
jobs:
  ci-tools:
    # CI 自身のテスト。一切の赤を許さない軽量ジョブ（必須チェック候補）
    if: ${{ !inputs.full }}
    runs-on: ubuntu-latest
    timeout-minutes: 5
    steps:
      - uses: actions/checkout@v4
        with: { persist-credentials: false }
      - uses: actions/setup-node@v4
        with: { node-version: 22 }
      - run: node --test tests/ci-select-tests.test.js tests/ci-baseline.test.js
  related:
    # 関連テスト（変更により FULL へ切替あり）。baseline 外の失敗のみ赤。既知の赤は warning で表示
    if: ${{ !inputs.full }}
    runs-on: ubuntu-latest
    timeout-minutes: 20
    steps:
      - uses: actions/checkout@v4
        with: { fetch-depth: 0, persist-credentials: false }
      - uses: actions/setup-node@v4
        with: { node-version: 22 }
      - run: test -f package-lock.json
      - run: npm ci --ignore-scripts
      - run: node tools/ci/select-tests.js --base origin/main --run --baseline tools/ci/baseline/pr29-run38190403528.json
  full:
    if: ${{ inputs.full }}
    runs-on: ubuntu-latest
    timeout-minutes: 30
    steps:
      - uses: actions/checkout@v4
        with: { persist-credentials: false }
      - uses: actions/setup-node@v4
        with: { node-version: 22 }
      - run: test -f package-lock.json
      - run: npm ci --ignore-scripts
      - name: Full tests (baseline なし。失敗は隠さない)
        run: npm test
