# 提案（未適用）: .github/workflows/pr-fast-tests.yml として人間が追加する。
# 無課金: GitHub-hosted runner 上の Node 標準テストのみ。外部APIなし。シークレット不使用。
name: PR fast tests
on:
  pull_request:
    branches: [main]
  workflow_dispatch:
    inputs:
      full:
        description: "全件テスト(npm test)を実行"
        type: boolean
        default: false
permissions:
  contents: read
concurrency:
  group: pr-fast-${{ github.ref }}
  cancel-in-progress: true
jobs:
  test:
    runs-on: ubuntu-latest
    timeout-minutes: 20
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0
      - uses: actions/setup-node@v4
        with:
          node-version: 22
      - run: npm ci --ignore-scripts || npm install --ignore-scripts
      - name: Related tests (PR)
        if: ${{ !inputs.full }}
        run: node tools/ci/select-tests.js --base origin/main --run
      - name: Full tests
        if: ${{ inputs.full }}
        run: npm test
      # ブラウザスモークは Playwright 導入を別承認とした後に有効化する（Phase B）。
